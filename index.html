        const corners = [
            {r: data[0], g: data[1], b: data[2]},

        ];
        const avgBg = {r:0,g:0,b:0};
        corners.forEach(c => { avgBg.r+=c.r; avgBg.g+=c.g; avgBg.b+=c.b; });

        const corners = [
            {r: data[0], g: data[1], b: data[2]},
            {r: data[(imageData.width-1)*4], g: data[(imageData.width-1)*4+1], b: data[(imageData.width-1)*4+2]},

        ];
        const avgBg = {r:0,g:0,b:0};
        corners.forEach(c => { avgBg.r+=c.r; avgBg.g+=c.g; avgBg.b+=c.b; });
        avgBg.r = Math.round(avgBg.r/4); avgBg.g = Math.round(avgBg.g/4); avgBg.b = Math.round(avgBg.b/4);
        for (let i = 0; i < data.length; i += 4) {

    function removeImageBackground(imageData, tolerance) {
        const data = imageData.data;
        const bgColor = {r: data[0], g: data[1], b: data[2]};
        const corners = [
            {r: data[0], g: data[1], b: data[2]},
            {r: data[(imageData.width-1)*4], g: data[(imageData.width-1)*4+1], b: data[(imageData.width-1)*4+2]},
            {r: data[(imageData.height-1)*imageData.width*4], g: data[(imageData.height-1)*imageData.width*4+1], b: data[(imageData.height-1)*imageData.width*4+2]},
            {r: data[((imageData.height-1)*imageData.width+imageData.width-1)*4], g: data[((imageData.height-1)*imageData.width+imageData.width-1)*4+1], b: data[((imageData.height-1)*imageData.width+imageData.width-1)*4+2]}
        ];
        const avgBg = {r:0,g:0,b:0};
        corners.forEach(c => { avgBg.r+=c.r; avgBg.g+=c.g; avgBg.b+=c.b; });
        avgBg.r = Math.round(avgBg.r/4); avgBg.g = Math.round(avgBg.g/4); avgBg.b = Math.round(avgBg.b/4);
        for (let i = 0; i < data.length; i += 4) {
            const dr = data[i]-avgBg.r, dg = data[i+1]-avgBg.g, db = data[i+2]-avgBg.b;
            const dist = Math.sqrt(dr*dr+dg*dg+db*db);
            if (dist < tolerance * 4.5) data[i+3] = 0;
        }
    }

    function removeImageBackground(imageData, tolerance) {
        const data = imageData.data;
        const bgColor = {r: data[0], g: data[1], b: data[2]};
        const corners = [
            {r: data[0], g: data[1], b: data[2]},
            {r: data[(imageData.width-1)*4], g: data[(imageData.width-1)*4+1], b: data[(imageData.width-1)*4+2]},
            {r: data[(imageData.height-1)*imageData.width*4], g: data[(imageData.height-1)*imageData.width*4+1], b: data[(imageData.height-1)*imageData.width*4+2]},
            {r: data[((imageData.height-1)*imageData.width+imageData.width-1)*4], g: data[((imageData.height-1)*imageData.width+imageData.width-1)*4+1], b: data[((imageData.height-1)*imageData.width+imageData.width-1)*4+2]}
        ];
        const avgBg = {r:0,g:0,b:0};
        corners.forEach(c => { avgBg.r+=c.r; avgBg.g+=c.g; avgBg.b+=c.b; });
        avgBg.r = Math.round(avgBg.r/4); avgBg.g = Math.round(avgBg.g/4); avgBg.b = Math.round(avgBg.b/4);
        for (let i = 0; i < data.length; i += 4) {
            const dr = data[i]-avgBg.r, dg = data[i+1]-avgBg.g, db = data[i+2]-avgBg.b;
            const dist = Math.sqrt(dr*dr+dg*dg+db*db);
            if (dist < tolerance * 4.5) data[i+3] = 0;
        }
    }

    function updateCutoutPreview() {
        if (!rawCropData) return;

        ctx.drawImage(rawCropData.canvas, 0, 0);
        const imageData = ctx.getImageData(0, 0, tempCanvas.width, tempCanvas.height);
        removeImageBackground(imageData, tolerance);
        ctx.putImageData(imageData, 0, 0);
        const previewCanvas = document.getElementById('cutoutPreviewCanvas');
        const scale = parseFloat(document.getElementById('cutoutScale').value) / 100;
        previewCanvas.width = rawCropData.width * scale;

        pctx.clearRect(0, 0, previewCanvas.width, previewCanvas.height);
        pctx.drawImage(tempCanvas, 0, 0, previewCanvas.width, previewCanvas.height);
    }
